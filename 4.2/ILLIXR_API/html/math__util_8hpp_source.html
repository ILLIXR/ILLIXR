21inline void projection(Eigen::Matrix4f* result, const float tan_left, const float tan_right, const float tan_up,
22 float const tan_down, const float near_z, const float far_z) {
23 const float tan_width = tan_right - tan_left;
24 const float tan_height = tan_up - tan_down;
25
26 // https://www.scratchapixel.com/lessons/3d-basic-rendering/perspective-and-orthographic-projection-matrix/building-basic-perspective-projection-matrix
27 (*result)(0, 0) = 2 / tan_width;
28 (*result)(0, 1) = 0;
29 (*result)(0, 2) = (tan_right + tan_left) / tan_width;
30 (*result)(0, 3) = 0;
31
32 (*result)(1, 0) = 0;
33 (*result)(1, 1) = 2 / tan_height;
34 (*result)(1, 2) = (tan_up + tan_down) / tan_height;
35 (*result)(1, 3) = 0;
36
37 (*result)(2, 0) = 0;
38 (*result)(2, 1) = 0;
39 (*result)(2, 2) = -far_z / (far_z - near_z);
40 (*result)(2, 3) = -(far_z * near_z) / (far_z - near_z);
41
42 (*result)(3, 0) = 0;
43 (*result)(3, 1) = 0;
44 (*result)(3, 2) = -1;
45 (*result)(3, 3) = 0;
46}

77inline void projection_fov(Eigen::Matrix4f* result, const float fov_left, const float fov_right, const float fov_up,

93inline void unreal_projection(Eigen::Matrix4f* result, const float fov_left, const float fov_right, const float fov_up,
94 const float fov_down) {
95 // Unreal uses a far plane at infinity and a near plane of 10 centimeters (0.1 meters)
96 constexpr float near_z = 0.1f;
97
98 const float angle_left = tanf(static_cast<float>(fov_left));
99 const float angle_right = tanf(static_cast<float>(fov_right));
100 const float angle_up = tanf(static_cast<float>(fov_up));
101 const float angle_down = tanf(static_cast<float>(fov_down));
102
103 const float sum_rl = angle_left + angle_right;
104 const float sum_tb = angle_up + angle_down;
105 const float inv_rl = 1.0f / (angle_right - angle_left);
106 const float inv_tb = 1.0f / (angle_up - angle_down);
107
108 (*result)(0, 0) = 2 * inv_rl;
109 (*result)(0, 1) = 0;
110 (*result)(0, 2) = sum_rl * (-inv_rl);
111 (*result)(0, 3) = 0;
112
113 (*result)(1, 0) = 0;
114 (*result)(1, 1) = 2 * inv_tb;
115 (*result)(1, 2) = sum_tb * (-inv_tb);
116 (*result)(1, 3) = 0;
117
118 (*result)(2, 0) = 0;
119 (*result)(2, 1) = 0;
120 (*result)(2, 2) = 0;
121 (*result)(2, 3) = near_z;
122
123 (*result)(3, 0) = 0;
124 (*result)(3, 1) = 0;
125 (*result)(3, 2) = -1;
126 (*result)(3, 3) = 0;
127}

130[[maybe_unused]] inline void godot_projection(Eigen::Matrix4f* result, const float fov_left, const float fov_right,
131 const float fov_up, const float fov_down) {
132 // Godot's default far and near planes are 4000m and 0.05m respectively.
133 // https://github.com/godotengine/godot/blob/e96ad5af98547df71b50c4c4695ac348638113e0/modules/openxr/openxr_util.cpp#L97
134 // The Vulkan implementation passes in GRAPHICS_OPENGL for some reason.
135 constexpr float near_z = 0.05f;
136 constexpr float far_z = 4000.f;
137 constexpr float offset_z = near_z;
138
139 const float angle_left = tanf(static_cast<float>(fov_left));
140 const float angle_right = tanf(static_cast<float>(fov_right));
141 const float angle_up = tanf(static_cast<float>(fov_up));
142 const float angle_down = tanf(static_cast<float>(fov_down));
143
144 const float angle_width = angle_right - angle_left;
145 const float angle_height = angle_up - angle_down;
146
147 Eigen::Matrix4f openxr_matrix;
148
149 openxr_matrix(0, 0) = 2 / angle_width;
150 openxr_matrix(0, 1) = 0;
151 openxr_matrix(0, 2) = (angle_right + angle_left) / angle_width;
152 openxr_matrix(0, 3) = 0;
153
154 openxr_matrix(1, 0) = 0;
155 openxr_matrix(1, 1) = 2 / angle_height;
156 openxr_matrix(1, 2) = (angle_up + angle_down) / angle_height;
157 openxr_matrix(1, 3) = 0;
158
159 openxr_matrix(2, 0) = 0;
160 openxr_matrix(2, 1) = 0;
161 openxr_matrix(2, 2) = -(far_z + offset_z) / (far_z - near_z);
162 openxr_matrix(2, 3) = -(far_z * (near_z + offset_z)) / (far_z - near_z);
163
164 openxr_matrix(3, 0) = 0;
165 openxr_matrix(3, 1) = 0;
166 openxr_matrix(3, 2) = -1;
167 openxr_matrix(3, 3) = 0;
168
169 // Godot then remaps the matrix...
170 // https://github.com/Khasehemwy/godot/blob/d950f5f83819240771aebb602bfdd4875363edce/core/math/projection.cpp#L722
171 Eigen::Matrix4f remap_z;
172
173 remap_z(0, 0) = 1;
174 remap_z(0, 1) = 0;
175 remap_z(0, 2) = 0;
176 remap_z(0, 3) = 0;
177
178 remap_z(1, 0) = 0;
179 remap_z(1, 1) = 1;
180 remap_z(1, 2) = 0;
181 remap_z(1, 3) = 0;
182
183 remap_z(2, 0) = 0;
184 remap_z(2, 1) = 0;
185 remap_z(2, 2) = -0.5;
186 remap_z(2, 3) = 0.5;
187
188 remap_z(3, 0) = 0;
189 remap_z(3, 1) = 0;
190 remap_z(3, 2) = 0;
191 remap_z(3, 3) = 1;
192
193 (*result) = remap_z * openxr_matrix;
194}

200inline Eigen::Matrix3f rotation(const float alpha, const float beta, const float gamma) {
201 Eigen::Matrix3f rot;
202 double ra = alpha * M_PI / 180.;
203 double rb = beta * M_PI / 180.;
204 double rg = gamma * M_PI / 180;
205 rot << static_cast<float>(cos(rg) * cos(rb)), static_cast<float>(cos(rg) * sin(rb) * sin(ra) - sin(rg) * cos(ra)),
206 static_cast<float>(cos(rg) * sin(rb) * cos(ra) + sin(rg) * sin(ra)), static_cast<float>(sin(rg) * cos(rb)),
207 static_cast<float>(sin(rg) * sin(rb) * sin(ra) + cos(rg) * cos(ra)),
208 static_cast<float>(sin(rg) * sin(rb) * cos(ra) - cos(rg) * sin(ra)), static_cast<float>(-sin(rb)),
209 static_cast<float>(cos(rb) * sin(ra)), static_cast<float>(cos(rb) * cos(ra));
210 return rot;
211}
