84// Convert a camera_intrinsics struct to a 1D numpy float32 array [fx, fy, cx, cy].
85// Copies the four floats by value — the struct is not contiguous in memory, so
86// zero-copy is not applicable here.
87static pybind11::array_t<float> intrinsics_to_numpy(const data_format::camera_intrinsics& intr) {
88 auto result = pybind11::array_t<float>(4);
89 auto buf = result.mutable_unchecked<1>();
90 buf(0) = intr.fx;
91 buf(1) = intr.fy;
92 buf(2) = intr.cx;
93 buf(3) = intr.cy;
94 return result;
95}

97// Wrap a DecodedFrameCache::Entry's RGB vector as a (H, W, 3) uint8 numpy
98// array without copying. The capsule keeps the Entry alive via a shared_ptr
99// to the cache entry data. Since Entry is owned by the cache (fixed array),
100// we share-own a copy of the rgb vector data instead.
101static pybind11::array_t<uint8_t> entry_to_numpy(const decode::decoded_frame_cache::Entry* entry) {
102 if (entry == nullptr || entry->rgb.empty())
103 return pybind11::array_t<uint8_t>();
104
105 // Copy the vector into a heap-allocated buffer owned by the capsule.
106 // This is one memcpy but avoids lifetime issues with the cache slot
107 // being overwritten while Python holds a reference.
108 auto* buf = new std::vector<uint8_t>(entry->rgb);
109 pybind11::capsule base(buf, [](void* p) {
110 delete static_cast<std::vector<uint8_t>*>(p);
111 });
112
113 return pybind11::array_t<uint8_t>({static_cast<pybind11::ssize_t>(entry->height),
114 static_cast<pybind11::ssize_t>(entry->width), static_cast<pybind11::ssize_t>(3)},
115 {static_cast<pybind11::ssize_t>(entry->width * 3), static_cast<pybind11::ssize_t>(3),
116 static_cast<pybind11::ssize_t>(1)},
117 buf->data(), base);
118}

124struct py_semantic_data_reader {
125 decode::decoded_frame_cache* cache;
126 decode::semantic_metadata_cache* metadata_cache;
127
128 [[nodiscard]] pybind11::object get() const {
129 auto entry_opt = cache ? cache->latest() : std::nullopt;
130 if (!entry_opt)
131 return pybind11::none();
132
133 auto meta_opt = metadata_cache ? metadata_cache->find(entry_opt->frame_number) : std::nullopt;
134 if (!meta_opt)
135 return pybind11::none();
136
137 // entry_opt->rgb is our own copy — safe to use without any lock.
138 // meta_opt->data is a shared_ptr — refcount keeps the frame alive.
139 auto val = meta_opt->data;
140
141 pybind11::dict data;
142 data["image"] = entry_to_numpy(&*entry_opt);
143 data["frame_number"] = entry_opt->frame_number;
144 data["image_width"] = val->intrinsics.width;
145 data["image_height"] = val->intrinsics.height;
146 data["depth"] = to_numpy_flat_safe(val, val->depth.data(), val->depth.size());
147 data["depth_width"] = val->depth_intrinsics.width;
148 data["depth_height"] = val->depth_intrinsics.height;
149 data["depth_near_z"] = val->depth_near_z;
150 data["intrinsics"] = intrinsics_to_numpy(val->intrinsics);
151 data["depth_intrinsics"] = intrinsics_to_numpy(val->depth_intrinsics);
152 data["rgb_camera_pose"] = to_numpy_4x4_safe(val, val->rgb_camera_pose);
153 data["depth_pose"] = to_numpy_4x4_safe(val, val->depth_pose);
154 data["max_depth_m"] = val->max_depth;
155 data["rgb_timestamp_ns"] = val->rgb_timestamp_ns;
156 data["depth_timestamp_ns"] = val->depth_timestamp_ns;
157
158 return data;
159 }
160};

166struct py_voice_query_reader {
167 switchboard::reader<data_format::semantic_xr::voice_query>* reader_;
168 uint64_t last_query_id_ = 0;
169
170 pybind11::object get() {
171 auto val = reader_->get_ro_nullable();
172 if (!val)
173 return pybind11::none();
174
175 // Only return if this is a new query we haven't delivered yet
176 if (val->query_id == last_query_id_)
177 return pybind11::none();
178
179 last_query_id_ = val->query_id;
180
181 pybind11::dict d;
182 d["query_id"] = val->query_id;
183 d["similarity_threshold"] = val->similarity_threshold;
184 d["min_match_similarity"] = val->min_match_similarity;
185
186 // PCM data — zero-copy with capsule
187 pybind11::capsule base(new std::shared_ptr<const data_format::semantic_xr::voice_query>(val), [](void* p) {
188 delete static_cast<std::shared_ptr<const data_format::semantic_xr::voice_query>*>(p);
189 });
190 d["pcm_data"] = pybind11::array_t<uint8_t>({static_cast<pybind11::ssize_t>(val->pcm_data.size())},
191 {static_cast<pybind11::ssize_t>(1)}, val->pcm_data.data(), base);
192
193 return d;
194 }
195};

212 void put(uint64_t query_id, pybind11::list point_clouds, pybind11::list colors, float server_latency,
213 const std::string& text_query) const {
214 auto resp = std::make_shared<data_format::semantic_xr::query_response>();
215 resp->query_id = query_id;
216 resp->server_query_processing = server_latency;
217 resp->text_query = text_query;
218 resp->num_point_clouds = static_cast<int32_t>(point_clouds.size());
219
220 resp->point_clouds.reserve(point_clouds.size());
221 for (auto& item : point_clouds) {
222 auto pc_dict = item.cast<pybind11::dict>();
223 data_format::semantic_xr::point_cloud pc;
224 pc.points = pc_dict["points"].cast<std::vector<float>>();
225 pc.num_points = static_cast<int32_t>(pc.points.size() / 3);
226 pc.centroid = pc_dict["centroid"].cast<std::vector<float>>();
227 resp->point_clouds.push_back(std::move(pc));
228 }
229
230 resp->colors = colors.cast<std::vector<float>>();
231
232 writer_->put(std::move(resp));
233 }
