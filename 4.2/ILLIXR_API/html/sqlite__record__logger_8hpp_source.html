128 void process(const std::vector<record>& record_batch, std::size_t batch_size) {
129 sqlite3pp::transaction xct{database_};
130 for (std::size_t i = 0; i < batch_size; ++i) {
131 // TODO(performance): reuse the sqlite3pp statement
132 // This currently has to be 'reinterpreted' for every iteration.
133 sqlite3pp::command cmd{database_, insert_str_.c_str()};
134 const record& r = record_batch[i];
135 for (int j = 0; j < static_cast<int>(record_header_.get_columns()); ++j) {
136 /*
137 If you get a `std::bad_any_cast` here, make sure the user didn't lie about record.get_record_header().
138 The types there should be the same as those in record.get_values().
139 */
140 const std::type_info& rh_type = record_header_.get_column_type(j);
141 if (rh_type == typeid(std::size_t)) {
142 cmd.bind(j + 1, static_cast<long long>(r.get_value<std::size_t>(j)));
143 } else if (rh_type == typeid(bool)) {
144 cmd.bind(j + 1, static_cast<long long>(r.get_value<bool>(j)));
145 } else if (rh_type == typeid(double)) {
146 cmd.bind(j + 1, r.get_value<double>(j));
147 } else if (rh_type == typeid(std::chrono::nanoseconds)) {
148 auto val = r.get_value<duration>(j);
149 cmd.bind(j + 1, static_cast<long long>(std::chrono::nanoseconds{val}.count()));
150 } else if (rh_type == typeid(std::chrono::high_resolution_clock::time_point)) {
151 auto val = r.get_value<std::chrono::high_resolution_clock::time_point>(j).time_since_epoch();
152 cmd.bind(j + 1, static_cast<long long>(std::chrono::nanoseconds{val}.count()));
153 } else if (rh_type == typeid(duration)) {
154 auto val = r.get_value<duration>(j);
155 cmd.bind(j + 1, static_cast<long long>(std::chrono::nanoseconds{val}.count()));
156 } else if (rh_type == typeid(time_point)) {
157 auto val = r.get_value<time_point>(j).time_since_epoch();
158 cmd.bind(j + 1, static_cast<long long>(std::chrono::nanoseconds{val}.count()));
159 } else if (rh_type == typeid(std::string)) {
160 // r.get_value<std::string>(j) returns a std::string temporary
161 // c_str() returns a pointer into that std::string temporary
162 // Therefore, need to copy.
163 cmd.bind(j + 1, r.get_value<std::string>(j).c_str(), sqlite3pp::copy);
164 } else {
165 throw std::runtime_error{std::string{"type "} + std::string{record_header_.get_column_type(j).name()} +
166 std::string{" not implemented"}};
167 }
168 }
169 RAC_ERRNO_MSG("sqlite_record_logger set errno before process cmd execute");
170
171 cmd.execute();
172 RAC_ERRNO_MSG("sqlite_record_logger after process cmd execute");
173 }
174 xct.commit();
175 }
