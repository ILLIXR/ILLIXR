41template<class Archive, typename T>
42static void load_sized_vector(Archive& ar, std::vector<T>& values, std::uint64_t maximum_size, const char* field_name) {
43 std::uint64_t size = 0;
44 ar >> size;
45 if (size > maximum_size || size > static_cast<std::uint64_t>(std::numeric_limits<std::size_t>::max())) {
46 throw std::runtime_error(std::string{"compressed_frame: invalid "} + field_name + " size");
47 }
48 values.resize(static_cast<std::size_t>(size));
49 if (size != 0) {
50 ar >> boost::serialization::make_array(values.data(), values.size());

136template<class Archive>
137static void load_packet(Archive& ar, AVPacket* pkt) {
138 int32_t pkt_size; // Use fixed-width type
139 ar >> pkt_size;
140 // FFmpeg's bitstream readers require zeroed AV_INPUT_BUFFER_PADDING_SIZE
141 // bytes beyond the payload. A payload-sized av_buffer_alloc is insufficient.
142 if (pkt_size < 0 || av_new_packet(pkt, pkt_size) < 0) {
143 throw std::runtime_error{"Failed to allocate received video packet"};
144 }
145 ar >> boost::serialization::make_array(pkt->data, pkt->size);
146 ar >> pkt->pts;
147 ar >> pkt->dts;
148 ar >> pkt->stream_index;
149 ar >> pkt->flags;
150 ar >> pkt->duration;
151 ar >> pkt->pos;
152 ar >> pkt->time_base.num;
153 ar >> pkt->time_base.den;
154 int32_t side_data_elems; // Use fixed-width type
155 ar >> side_data_elems;
156 if (side_data_elems < 0) {
157 throw std::runtime_error{"Invalid video packet side-data count"};
158 }
159 for (int i = 0; i < side_data_elems; i++) {
160 int32_t side_data_type;
161 uint64_t side_data_size;
162 ar >> side_data_type;
163 ar >> side_data_size;
164 auto* data = av_packet_new_side_data(pkt, static_cast<AVPacketSideDataType>(side_data_type),
165 static_cast<size_t>(side_data_size));
166 if (data == nullptr) {
167 throw std::runtime_error{"Failed to allocate received video packet side data"};
168 }
169 ar >> boost::serialization::make_array(data, static_cast<size_t>(side_data_size));
170 }
171}
