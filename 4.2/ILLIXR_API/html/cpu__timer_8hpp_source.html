28static int illixr_clock_gettime(clockid_t clock_id, timespec* ts) {
29 if (clock_id == CLOCK_MONOTINIC) {
30 static LARGE_INTEGER start_ticks;
31 static LARGE_INTEGER frequency;
32 static bool initialized = false;
33
34 if (!initialized) {
35 QueryPerformanceFrequency(&frequency);
36 QueryPerformanceCounter(&start_ticks);
37 initialized = true;
38 }
39
40 LARGE_INTEGER current_ticks;
41 QueryPerformanceCounter(&current_ticks);
42
43 LONGLONG diff = current_ticks.QuadPart - start_ticks.QuadPart;
44 ts->tv_sec = diff / frequency.QuadPart;
45 ts->tv_nsec = static_cast<long>(((diff % frequency.QuadPart) * 1000000000) / frequency.QuadPart);
46 return 0;
47 } else if (clock_id == CLOCK_THREAD_CPUTIME_ID) {
48 FILETIME creation_time, exit_time, kernel_time, user_time;
49 if (!GetThreadTimes(GetCurrentThread(), &creation_time, &exit_time, &kernel_time, &user_time))
50 return -1;
51
52 ULARGE_INTEGER ktime, utime;
53 ktime.LowPart = kernel_time.dwLowDateTime;
54 ktime.HighPart = kernel_time.dwHighDateTime;
55
56 utime.LowPart = user_time.dwLowDateTime;
57 utime.HighPart = user_time.dwHighDateTime;
58
59 uint64_t total_time_100ns = ktime.QuadPart + utime.QuadPart;
60
61 // convert to ns
62 uint64_t total_time_ns = total_time_100ns * 100;
63 ts->tv_sec = static_cast<time_t>(total_time_ns / 1000000000ULL);
64 ts->tv_nsec = static_cast<time_t>(total_time_ns % 1000000000ULL);
65 return 0;
66 }
67 return -1;
68}
