142state_plus predict_mean_rk4(double dt, const state_plus& sp, const Eigen::Vector3d& ang_vel, const Eigen::Vector3d& linear_acc,
143 const Eigen::Vector3d& ang_vel2, const Eigen::Vector3d& linear_acc2) {
144 Eigen::Vector3d av = ang_vel;
145 Eigen::Vector3d la = linear_acc;
146 const Eigen::Vector3d delta_av = (ang_vel2 - ang_vel) / dt;
147 const Eigen::Vector3d delta_la = (linear_acc2 - linear_acc) / dt;
148
149 // y0 ================
150 data_format::proper_quaterniond q_0 = sp.orientation; // initial orientation quaternion
151 Eigen::Vector3d p_0 = sp.position; // initial position vector
152 Eigen::Vector3d v_0 = sp.velocity; // initial velocity vector
153
154 // Calculate the RK4 coefficients
155 // solve orientation
156 // k1
157 data_format::proper_quaterniond k1_q = q_dot(av, dq_0) * dt;
158 av += 0.5 * delta_av * dt;
159 // k2
160 data_format::proper_quaterniond dq_1 = delta_q(k1_q);
161 data_format::proper_quaterniond k2_q = q_dot(av, dq_1) * dt;
162 // k3
163 data_format::proper_quaterniond dq_2 = delta_q(k2_q);
164 data_format::proper_quaterniond k3_q = q_dot(av, dq_2) * dt;
165 // k4
166 av += 0.5 * delta_av * dt;
167 data_format::proper_quaterniond dq_3 = delta_q(2. * k3_q);
168 data_format::proper_quaterniond k4_q = q_dot(av, dq_3) * dt;
169
170 // solve velocity
171 // k1
172 Eigen::Vector3d k1_v = v_dot(dq_0, q_0, la) * dt;
173 // k2
174 la += 0.5 * delta_la * dt;
175 Eigen::Vector3d k2_v = v_dot(dq_1, q_0, la) * dt;
176 // k3
177 Eigen::Vector3d k3_v = v_dot(dq_2, q_0, la) * dt;
178 // k4
179 la += 0.5 * delta_la * dt;
180 Eigen::Vector3d k4_v = v_dot(dq_3, q_0, la) * dt;
181
182 // solve position
183 // k1
184 Eigen::Vector3d k1_p = v_0 * dt;
185 // k2
186 Eigen::Vector3d k2_p = p_dot(v_0, k1_v) * dt;
187 // k3
188 Eigen::Vector3d k3_p = p_dot(v_0, k2_v) * dt;
189 // k4
190 Eigen::Vector3d k4_p = p_dot(v_0, 2. * k3_v) * dt;
191
192 // y+dt ================
193 state_plus state_plus;
194 data_format::proper_quaterniond dq = solve(dq_0, k1_q, k2_q, k3_q, k4_q);
195 dq.normalize();
196 state_plus.orientation = q_0 * dq;
197 state_plus.orientation.normalize();
198 state_plus.position = solve(p_0, k1_p, k2_p, k3_p, k4_p);
199 state_plus.velocity = solve(v_0, k1_v, k2_v, k3_v, k4_v);
200 return state_plus;
201}
