289 void log(const record& r) {
290 if (logger_) {
291 buffer_.push_back(r);
292 // Log coalescer should only be used with
293 // In the common case, they will be the same pointer, quickly check the pointers.
294 // In the less common case, we check for object-structural equality.
295#ifndef NDEBUG
296 if (&r.get_record_header() != &buffer_[0].get_record_header() &&
297 r.get_record_header() == buffer_[0].get_record_header()) {
298 spdlog::get("illixr")->error("[record_logger] Tried to push a record of type {} to a record logger for type {}",
299 r.get_record_header().to_string(), buffer_[0].get_record_header().to_string());
300 abort();
301 }
302#endif
303 maybe_flush();
304 }
305 }
306
310 void maybe_flush() {
311 if (std::chrono::high_resolution_clock::now() > last_log_ + LOG_BUFFER_DELAY) {
312 flush();
313 }
314 }
315
319 void flush() {
320 if (logger_) {
321 std::vector<record> buffer2;
322 buffer_.swap(buffer2);
323 logger_->log(buffer2);
324 last_log_ = std::chrono::high_resolution_clock::now();
325 }
326 }

329 return bool(logger_);
330 }
