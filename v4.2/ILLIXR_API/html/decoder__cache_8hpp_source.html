15// ---------------------------------------------------------------------------
16// decoded_frame_cache
17//
18// Ring buffer storing decoded RGB frames alongside their frame numbers.
19// Sized to match the switchboard topic history depth (256).
20//
21// Written by the decode callback (switchboard thread).
22// Read by latest() (Python thread).
23// Protected by a mutex.
24//
25// Each slot owns a std::vector<uint8_t> that holds a copy of the pinned
26// host RGB buffer. The copy is necessary because NvdecDecoder reuses its
27// pinned buffer on each decode() call.
28// ---------------------------------------------------------------------------
29class decoded_frame_cache {
30public:
31 static constexpr size_t CAPACITY = 256;
32
33 struct Entry {
34 std::vector<uint8_t> rgb; // packed RGB uint8, H*W*3 bytes
35 int32_t frame_number = -1;
36 int32_t width = 0;
37 int32_t height = 0;
38 bool valid = false;
39 };
40
41 // Store a decoded RGB frame. Copies from the pinned host pointer.
42 void store(int32_t frame_number, int32_t w, int32_t h, const uint8_t* rgb_data) {
43 std::lock_guard<std::mutex> lock(mutex_);
44 Entry& slot = entries_[next_slot_];
45 slot.frame_number = frame_number;
46 slot.width = w;
47 slot.height = h;
48 const auto nbytes = static_cast<size_t>(w * h * 3);
49 slot.rgb.resize(nbytes);
50 std::memcpy(slot.rgb.data(), rgb_data, nbytes);
51 slot.valid = true;
52 next_slot_ = (next_slot_ + 1) % CAPACITY;
53 }
54
55 // Return a copy of the most recently stored entry, taken under the lock.
56 // Returns nullopt if no frame has been stored yet.
57 std::optional<Entry> latest() const {
58 std::lock_guard<std::mutex> lock(mutex_);
59 int32_t best_fn = -1;
60 const Entry* best = nullptr;
61 for (const auto& e : entries_) {
62 if (e.valid && e.frame_number > best_fn) {
63 best_fn = e.frame_number;
64 best = &e;
65 }
66 }
67 if (best == nullptr)
68 return std::nullopt;
69 return *best; // copy made while lock is held
70 }
71
72private:
73 mutable std::mutex mutex_;
74 std::array<Entry, CAPACITY> entries_;
75 size_t next_slot_ = 0;
76};

78// ---------------------------------------------------------------------------
79// semantic_metadata_cache
80//
81// Ring buffer storing each semantic_frame (depth, poses, intrinsics, etc.),
82// keyed by frame_number. Holds a switchboard::ptr to the frame itself rather
83// than copying individual fields out, so the frame's other data stays alive
84// and zero-copy-accessible for as long as the slot is valid.
85//
86// decode lags arrival by a variable amount (see nvdec_decoder::decode), so
87// the frame most recently decoded is not the frame most recently arrived.
88// This cache lets get() look up the arrival-time data (depth/pose/etc.)
89// for whichever frame_number the decoder actually finished decoding,
90// rather than assuming the two arrive together.
91//
92// Written by on_semantic_data() (switchboard thread), at arrival, before
93// the decode call. Read by find() (Python thread).
94// Protected by a mutex.
95// ---------------------------------------------------------------------------
96class semantic_metadata_cache {
97public:
98 static constexpr size_t CAPACITY = 256;
99
100 struct Entry {
101 switchboard::ptr<const data_format::semantic_frame> data;
102 int32_t frame_number = -1;
103 bool valid = false;
104 };
105
106 // Store a frame's metadata, keyed by frame_number.
107 void store(int32_t frame_number, const switchboard::ptr<const data_format::semantic_frame>& frame) {
108 std::lock_guard<std::mutex> lock(mutex_);
109 Entry& slot = entries_[next_slot_];
110 slot.data = frame;
111 slot.frame_number = frame_number;
112 slot.valid = true;
113 next_slot_ = (next_slot_ + 1) % CAPACITY;
114 }
115
116 // Return a copy of the entry matching frame_number, taken under the lock.
117 // Returns nullopt if not found (evicted or not yet arrived).
118 std::optional<Entry> find(int32_t frame_number) const {
119 std::lock_guard<std::mutex> lock(mutex_);
120 for (const auto& e : entries_) {
121 if (e.valid && e.frame_number == frame_number)
122 return e; // copy made while lock is held
123 }
124 return std::nullopt;
125 }
126
127private:
128 mutable std::mutex mutex_;
129 std::array<Entry, CAPACITY> entries_;
130 size_t next_slot_ = 0;
131};
