139 void update(std::array<XrHandJointLocationEXT, XR_HAND_JOINT_COUNT_EXT>& pose,
140 std::array<XrHandJointVelocityEXT, XR_HAND_JOINT_COUNT_EXT>& velocity) {
141 uint8_t tracked_count = 0;
142 for (uint32_t j = 0; j < XR_HAND_JOINT_COUNT_EXT; j++) {
143 joints[j] = {pose[j], velocity[j]};
144 if (pose[j].locationFlags & XR_SPACE_LOCATION_ORIENTATION_VALID_BIT &&
145 pose[j].locationFlags & XR_SPACE_LOCATION_POSITION_VALID_BIT)
146 tracked_count++;
147 }
148 is_active = (static_cast<float>(tracked_count) / static_cast<float>(XR_HAND_JOINT_COUNT_EXT)) > 0.;
149 }
