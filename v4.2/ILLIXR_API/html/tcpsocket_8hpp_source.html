83 // regardless of whether the caller remembered to set these options first.

138 // Allow the reuse of local addresses.
139 // On Linux/Android, also sets SO_REUSEPORT so that a new socket can bind
140 // to a port still in TIME_WAIT after a crash, without waiting ~60 seconds.
141 void socket_set_reuseaddr() const {
142#if defined(_WIN32) || defined(_WIN64)
143 int enable = 1;
144 if (setsockopt(fd_, SOL_SOCKET, SO_REUSEADDR, (char*) &enable, sizeof(enable)) < 0) {
145 int err = WSAGetLastError();
146 throw std::runtime_error("Reuseaddr failed");
147 }
148#else
149 const int enable = 1;
150 setsockopt(fd_, SOL_SOCKET, SO_REUSEADDR, &enable, sizeof(int));
151 setsockopt(fd_, SOL_SOCKET, SO_REUSEPORT, &enable, sizeof(int));
152#endif
153 }

155 // Set SO_LINGER with a zero timeout so that close() sends a TCP RST
156 // rather than going through the graceful FIN/FIN-ACK sequence.
157 // This prevents the socket from entering TIME_WAIT after a crash,
158 // allowing the port to be reused immediately on restart.
159 void socket_set_linger_zero() const {
160#if defined(_WIN32) || defined(_WIN64)
161 LINGER lg{};
162 lg.l_onoff = 1;
163 lg.l_linger = 0;
164 setsockopt(fd_, SOL_SOCKET, SO_LINGER, reinterpret_cast<const char*>(&lg), sizeof(lg));
165#else
166 struct linger lg{};
167 lg.l_onoff = 1;
168 lg.l_linger = 0;
169 setsockopt(fd_, SOL_SOCKET, SO_LINGER, &lg, sizeof(lg));
170#endif
171 }

265 [[nodiscard]] std::string::const_iterator write_helper(const std::string::const_iterator& begin,
266 const std::string::const_iterator& end) const {
267 BYTE_TYPE bytes_written =
268#if defined(_WIN32) || defined(_WIN64)
269 send(fd_, &*begin, static_cast<int>(end - begin), 0);
270 if (bytes_written < 0)
271 throw std::runtime_error("Write failed");
272#else
273 send(fd_, &*begin, end - begin, MSG_NOSIGNAL);
274 if (bytes_written < 0) {
275 if (errno == EINTR) {
276 return begin;
277 }
278 throw std::runtime_error(std::string{"Write failed: "} + std::strerror(errno));
279 }
280#endif
281 return begin + bytes_written;
282 }
