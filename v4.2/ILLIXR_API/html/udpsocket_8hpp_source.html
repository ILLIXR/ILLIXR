67 // Bind to a local address and port (required on the server side, optional on the client)
68 void socket_bind(const std::string& ip, int port) const {
69 sockaddr_in local_addr{};
70 local_addr.sin_family = AF_INET;
71 local_addr.sin_port = htons(static_cast<uint16_t>(port));
72 local_addr.sin_addr.s_addr = ip.empty() ? INADDR_ANY : inet_addr(ip.c_str());
73 if (bind(fd_, reinterpret_cast<sockaddr*>(&local_addr), sizeof(local_addr)) < 0)
74 throw std::runtime_error("UDP bind failed");
75 }

121 // Store the peer address for use by write_data().
122 // On the client this is called once with the server's address.
123 // On the server this is called each time a datagram is received from a new peer.
124 void set_peer(const std::string& ip, int port) {
125 std::lock_guard<std::mutex> lock(peer_mutex_);
126 std::memset(&peer_addr_, 0, sizeof(peer_addr_));
127 peer_addr_.sin_family = AF_INET;
128 peer_addr_.sin_port = htons(static_cast<uint16_t>(port));
129 inet_pton(AF_INET, ip.c_str(), &peer_addr_.sin_addr);
130 peer_set_ = true;
131 }

133 // Store the peer address directly from a sockaddr_in, e.g. the src_addr filled in
134 // by read_data(). This lets the server learn the client's address from an incoming
135 // datagram, since UDP is connectionless and the server has no peer until it hears
136 // from one.
137 void set_peer(const sockaddr_in& addr) {
138 std::lock_guard<std::mutex> lock(peer_mutex_);
139 peer_addr_ = addr;
140 peer_set_ = true;

143 // Send a datagram to the previously set peer address.
144 // Returns false if no peer has been set or the send fails.
145 bool write_data(const std::string& buffer) const {
146 std::lock_guard<std::mutex> lock(peer_mutex_);
147 if (!peer_set_)
148 return false;
149 BYTE_TYPE sent = sendto(fd_,
150#if defined(_WIN32) || defined(_WIN64)
151 buffer.data(), static_cast<int>(buffer.size()),
152#else
153 buffer.data(), buffer.size(),
154#endif
155 0, reinterpret_cast<const sockaddr*>(&peer_addr_), sizeof(peer_addr_));
156 return sent == static_cast<BYTE_TYPE>(buffer.size());
157 }

171 // Receive a single datagram. Blocks until data arrives.
172 // Optionally fills src_addr with the sender's address.
173 [[nodiscard]] std::string read_data(sockaddr_in* src_addr = nullptr) const {
174 char buffer[BUFFER_SIZE];
175 sockaddr_in from{};
176#if defined(_WIN32) || defined(_WIN64)
177 int from_len = sizeof(from);
178#else
179 socklen_t from_len = sizeof(from);
180#endif
181 BYTE_TYPE bytes = recvfrom(fd_, buffer,
182#if defined(_WIN32) || defined(_WIN64)
183 static_cast<int>(BUFFER_SIZE),
184#else
185 BUFFER_SIZE,
186#endif
187 0, reinterpret_cast<sockaddr*>(&from), &from_len);
188 if (bytes <= 0)
189 return {};
190 if (src_addr != nullptr)
191 *src_addr = from;
192 return std::string(buffer, static_cast<size_t>(bytes));
193 }

196 [[nodiscard]] std::string local_address() const {

200#else
201 socklen_t size = sizeof(local);

207#else
208 char* ip = inet_ntoa(local.sin_addr);
209#endif
210 return std::string(ip) + ":" + std::to_string(ntohs(local.sin_port));
