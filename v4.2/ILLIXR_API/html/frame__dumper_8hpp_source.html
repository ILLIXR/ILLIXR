39 // Dump raw NV12 data to a file
40 void dump_nv12(const uint8_t* data, int width, int height, int eye, uint64_t frame_num) {
41 if (dump_count_ >= max_dumps_)

98 // Dump the current framebuffer contents
99 void dump_framebuffer(int width, int height, int eye, uint64_t frame_num) {
100 if (dump_count_ >= max_dumps_)
101 return;
102
103 std::vector<uint8_t> pixels(width * height * 4); // RGBA
104 glReadPixels(0, 0, width, height, GL_RGBA, GL_UNSIGNED_BYTE, pixels.data());
105
106 GLenum err = glGetError();
107 if (err != GL_NO_ERROR) {
108 spdlog::get("illixr")->error("frame_dumper: glReadPixels failed: 0x{:X}", err);
109 return;
110 }
111
112 // Convert RGBA to RGB for PPM
113 std::vector<uint8_t> rgb(width * height * 3);
114 for (int i = 0; i < width * height; i++) {
115 rgb[i * 3 + 0] = pixels[i * 4 + 0];
116 rgb[i * 3 + 1] = pixels[i * 4 + 1];
117 rgb[i * 3 + 2] = pixels[i * 4 + 2];
118 }
119
120 std::string filename = dump_dir_ + "/fb_" + std::to_string(frame_num) + "_eye" + std::to_string(eye) + ".ppm";
121
122 std::ofstream file(filename, std::ios::binary);
123 if (file.is_open()) {
124 file << "P6\n" << width << " " << height << "\n255\n";
125 file.write(reinterpret_cast<const char*>(rgb.data()), rgb.size());
126 file.close();
127 spdlog::get("illixr")->info("frame_dumper: Wrote framebuffer {}", filename);
128 dump_count_++;
129 }

166 void nv12_to_rgb(const uint8_t* nv12, uint8_t* rgb, int width, int height) {
167 const uint8_t* y_plane = nv12;
168 const uint8_t* uv_plane = nv12 + width * height;
169
170 for (int j = 0; j < height; j++) {
171 for (int i = 0; i < width; i++) {
172 int y_idx = j * width + i;
173 int uv_idx = (j / 2) * width + (i & ~1); // UV is half resolution, interleaved
174
175 int Y = y_plane[y_idx];
176 int U = uv_plane[uv_idx];
177 int V = uv_plane[uv_idx + 1];
178
179 // BT.601 conversion (commonly used)
180 int C = Y - 16;
181 int D = U - 128;
182 int E = V - 128;
183
184 int R = (298 * C + 409 * E + 128) >> 8;
185 int G = (298 * C - 100 * D - 208 * E + 128) >> 8;
186 int B = (298 * C + 516 * D + 128) >> 8;
187
188 // Clamp
189 R = R < 0 ? 0 : (R > 255 ? 255 : R);
190 G = G < 0 ? 0 : (G > 255 ? 255 : G);
191 B = B < 0 ? 0 : (B > 255 ? 255 : B);
192
193 int rgb_idx = (j * width + i) * 3;
194 rgb[rgb_idx + 0] = static_cast<uint8_t>(R);
195 rgb[rgb_idx + 1] = static_cast<uint8_t>(G);
196 rgb[rgb_idx + 2] = static_cast<uint8_t>(B);
197 }
198 }

201 void dump_regular_texture(GLuint texture_id, int width, int height, int eye, uint64_t frame_num) {
202 // Create FBO to read from texture
203 GLuint fbo;
204 glGenFramebuffers(1, &fbo);
205 glBindFramebuffer(GL_FRAMEBUFFER, fbo);
206 glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D, texture_id, 0);
207
208 if (glCheckFramebufferStatus(GL_FRAMEBUFFER) == GL_FRAMEBUFFER_COMPLETE) {
209 dump_framebuffer(width, height, eye, frame_num);

211 spdlog::get("illixr")->error("frame_dumper: Framebuffer incomplete for texture read");
212 }
213
214 glBindFramebuffer(GL_FRAMEBUFFER, 0);
215 glDeleteFramebuffers(1, &fbo);
216 }

234// Get the app's external files directory (no permissions needed)
235inline std::string get_app_files_dir(android_app* app) {
236 if (!app || !app->activity) {
237 return "";
238 }
239
240 // Get JNI env
241 JNIEnv* env = nullptr;
242 app->activity->vm->AttachCurrentThread(&env, nullptr);
243 if (!env) {
244 return "";
245 }
246
247 // Get activity class
248 jclass activity_class = env->GetObjectClass(app->activity->clazz);
249
250 // Get getExternalFilesDir method
251 jmethodID get_external_files_dir =
252 env->GetMethodID(activity_class, "getExternalFilesDir", "(Ljava/lang/String;)Ljava/io/File;");
253
254 // Call getExternalFilesDir(null)
255 jobject file_obj = env->CallObjectMethod(app->activity->clazz, get_external_files_dir, nullptr);
256
257 if (!file_obj) {
258 env->DeleteLocalRef(activity_class);
259 return "";
260 }
261
262 // Get File.getAbsolutePath()
263 jclass file_class = env->GetObjectClass(file_obj);
264 jmethodID get_path = env->GetMethodID(file_class, "getAbsolutePath", "()Ljava/lang/String;");
265 jstring path_str = (jstring) env->CallObjectMethod(file_obj, get_path);
266
267 const char* path_chars = env->GetStringUTFChars(path_str, nullptr);
268 std::string result(path_chars);
269
270 env->ReleaseStringUTFChars(path_str, path_chars);
271 env->DeleteLocalRef(path_str);
272 env->DeleteLocalRef(file_class);
273 env->DeleteLocalRef(file_obj);
274 env->DeleteLocalRef(activity_class);
275
276 return result;
277}
