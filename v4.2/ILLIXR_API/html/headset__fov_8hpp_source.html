34// Select before starting the process. Keep Quest 3 as the fallback for existing
35// Android launchers; the desktop Index launchers explicitly set ILLIXR_HEADSET.
36inline std::string configured_headset() {
37 const char* value = std::getenv("ILLIXR_HEADSET");
38 const std::string headset = value == nullptr ? "quest3" : value;
39 if (headset != "quest3" && headset != "index") {
40 throw std::runtime_error("Invalid ILLIXR_HEADSET='" + headset + "'; expected quest3 or index");
41 }
42 return headset;
43}
